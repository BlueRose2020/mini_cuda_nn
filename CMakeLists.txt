cmake_minimum_required(VERSION 3.18)
project(mini_cuda_nn LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(MINI_CUDA "Build CUDA backend" OFF)
add_library(mininn src/tensor.cpp src/ops.cpp src/autograd.cpp src/optimizer.cpp src/device.cpp)
target_include_directories(mininn PUBLIC include PRIVATE src)
if(MINI_CUDA)
 enable_language(CUDA)
 add_library(mininn_cuda cuda/kernels.cu)
 target_compile_definitions(mininn PUBLIC MINI_CUDA=1)
 target_link_libraries(mininn PUBLIC mininn_cuda)
endif()
add_executable(train examples/train.cpp)
target_link_libraries(train PRIVATE mininn)
